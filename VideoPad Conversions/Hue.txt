/*
=========================
VideoPad Hue by Jimmyhogs
=========================
KFLOAT0 = Hue (0 to 360)
=========================
*/

kReadIndex = (float2)(x, y);
kRead();

float r = kOutput.x;
float g = kOutput.y;
float b = kOutput.z;

float hue = clamp(kFloat0, 0.0f, 360.0f);

float nr = r;
float ng = g;
float nb = b;

if (hue <= 120.0f)
{
    float t = hue / 120.0f;

    nr = r * (1.0f - t) + b * t;
    ng = g * (1.0f - t) + r * t;
    nb = b * (1.0f - t) + g * t;
}

else if (hue <= 240.0f)
{
    float t = (hue - 120.0f) / 120.0f;

    nr = b * (1.0f - t) + g * t;
    ng = r * (1.0f - t) + b * t;
    nb = g * (1.0f - t) + r * t;
}

else
{
    float t = (hue - 240.0f) / 120.0f;

    nr = g * (1.0f - t) + r * t;
    ng = b * (1.0f - t) + g * t;
    nb = r * (1.0f - t) + b * t;
}

kOutput.xyz = clamp(
    (float3)(nr, ng, nb),
    0.0f,
    1.0f
);
